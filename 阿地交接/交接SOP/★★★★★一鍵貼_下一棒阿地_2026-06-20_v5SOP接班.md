```js
// 設日期 + 小時 8 + 分鐘 0

  .find(i => i.placeholder === '年-月-日');

const hourBtn = Array.from(document.querySelectorAll('[role="spinbutton"]'))
  .find(s => s.getAttribute('aria-label') === '小時');
const cur_h = parseInt(hourBtn.getAttribute('aria-valuenow'));
const diff_h = cur_h - 8;
hourBtn.focus();
for (let i = 0; i < Math.abs(diff_h); i++) {
  const dir = diff_h > 0 ? 'ArrowDown' : 'ArrowUp';
  const kc = diff_h > 0 ? 40 : 38;
  hourBtn.dispatchEvent(new KeyboardEvent('keydown', { key: dir, code: dir, keyCode: kc, which: kc, bubbles: true, cancelable: true }));
  hourBtn.dispatchEvent(new KeyboardEvent('keyup', { key: dir, code: dir, keyCode: kc, which: kc, bubbles: true, cancelable: true }));
}

// 改分鐘 → 0(同樣邏輯)
const minBtn = Array.from(document.querySelectorAll('[role="spinbutton"]'))
  .find(s => s.getAttribute('aria-label') === '分鐘');

const diff_m = cur_m - 0;
minBtn.focus();
for (let i = 0; i < Math.abs(diff_m); i++) {
  const dir = diff_m > 0 ? 'ArrowDown' : 'ArrowUp';
  const kc = diff_m > 0 ? 40 : 38;
  minBtn.dispatchEvent(new KeyboardEvent('keydown', { key: dir, code: dir, keyCode: kc, which: kc, bubbles: true, cancelable: true }));
```

### Step 9 — 點「排定時間」+ ★ v5 poll 法等推銷 dialog

```js
const btn = Array.from(document.querySelectorAll('button, [role="button"]'))
```

```js
// ★ v5 poll 法:0.5s × 20 主動掃,不固定 wait 5s
for (let i = 0; i < 20; i++) {
  await new Promise(r => setTimeout(r, 500));
  const dialog = document.querySelector('[role="dialog"]');
  if (dialog) {
    const later = Array.from(dialog.querySelectorAll('button, [role="button"]'))
      .filter(b => b.offsetParent)
      .find(b => (b.textContent || '').trim() === '稍後再說');
    if (later) { later.click(); return { iter: i, closed: true }; }
  }
  if (location.href.includes('content_calendar')) return { iter: i, url_changed: true };
}
```
